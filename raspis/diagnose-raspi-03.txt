#!/bin/bash
URL="https://timechainmap.com/map/"
ZOOM="1"
BROWSER="/usr/bin/chromium"
K="/home/admin/kiosk"
PROFILE="$K/profile"
exec > "$K/start.log" 2>&1
echo "$(date) Start"

# 1) Warten bis die Uhrzeit per Internet gestellt ist (max. 60 s)
for i in $(seq 1 60); do
    [ "$(timedatectl show -p NTPSynchronized --value 2>/dev/null)" = "yes" ] && break
    sleep 1
done
echo "$(date) Uhrzeit: $(timedatectl show -p NTPSynchronized --value 2>/dev/null)"

# 2) Warten bis die Website antwortet (max. 60 s)
for i in $(seq 1 20); do
    curl -s -o /dev/null --max-time 3 "$URL" && break
    sleep 1
done
echo "$(date) Netz-Check fertig"
sleep 3

# 3) Helfer starten (Scrollbalken/Maus weg, Auto-F5)
pkill -f "$K/helper.py" 2>/dev/null
python3 "$K/helper.py" > "$K/helper.log" 2>&1 &

# 4) Chromium starten - und neu starten, falls er abstuerzt
while true; do
    # "Chromium wurde nicht richtig beendet"-Leiste verhindern
    PREF="$PROFILE/Default/Preferences"
    [ -f "$PREF" ] && sed -i 's/"exited_cleanly":false/"exited_cleanly":true/; s/"exit_type":"[^"]*"/"exit_type":"Normal"/' "$PREF"

    echo "$(date) Chromium startet"
    "$BROWSER" --kiosk \
        --user-data-dir="$PROFILE" \
        --remote-debugging-port=9222 \
        --hide-scrollbars \
        --force-device-scale-factor="$ZOOM" \
        --noerrdialogs --disable-infobars --disable-session-crashed-bubble \
        --no-first-run --password-store=basic \
        --check-for-update-interval=31536000 \
        "$URL"
    echo "$(date) Chromium beendet (Code $?)"
    sleep 5
done
